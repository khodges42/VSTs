cmake_minimum_required(VERSION 3.25)
project(dubmist98 VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE CLOUDS CONFIGURE_DEPENDS vendor/clouds/dsp/*.cc)
add_library(dubmist_core ${CLOUDS} vendor/clouds/resources.cc vendor/stmlib/dsp/units.cc vendor/stmlib/dsp/atan.cc vendor/stmlib/utils/random.cc src/engine.cpp src/beat_repeat.cpp)
target_include_directories(dubmist_core PUBLIC src vendor)
target_compile_definitions(dubmist_core PUBLIC TEST _USE_MATH_DEFINES)
if(MSVC)
 target_compile_options(dubmist_core PUBLIC /permissive- /utf-8 /FIcstdio /wd4244 /wd4267 /wd4305)
endif()
set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/dubmist-deps/JUCE" CACHE PATH "Pinned JUCE 7.0.12")
if(NOT EXISTS "${JUCE_SOURCE_DIR}/CMakeLists.txt")
 message(FATAL_ERROR "Run scripts/bootstrap.ps1 to obtain pinned JUCE")
endif()
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/juce")
juce_add_plugin(Dubmist98 COMPANY_NAME "Voidmutt Enterprises" BUNDLE_ID "org.voidmutt.dubmist98"
 PLUGIN_MANUFACTURER_CODE Vmtt PLUGIN_CODE Dm98 PRODUCT_NAME "Dubmist 98"
 DESCRIPTION "Dubmist 98 Enterprise Edition SE" VERSION 0.2.0
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Granular FORMATS VST3)
target_sources(Dubmist98 PRIVATE plugin/Processor.cpp plugin/Editor.cpp)
target_include_directories(Dubmist98 PUBLIC plugin)
target_compile_definitions(Dubmist98 PUBLIC JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Dubmist98 PUBLIC dubmist_core juce::juce_audio_utils PRIVATE juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(dubmist_validate PRODUCT_NAME dubmist_validate)
target_sources(dubmist_validate PRIVATE tests/validate.cpp)
target_link_libraries(dubmist_validate PRIVATE Dubmist98)
add_test(NAME clouds_state_audio_reference COMMAND dubmist_validate "${CMAKE_CURRENT_BINARY_DIR}/validation")
add_executable(dubmist_realtime tests/realtime.cpp)
target_link_libraries(dubmist_realtime PRIVATE dubmist_core)
add_test(NAME clouds_realtime COMMAND dubmist_realtime)
juce_add_console_app(dubmist_host PRODUCT_NAME dubmist_host)
target_sources(dubmist_host PRIVATE tests/vst3_host.cpp)
target_compile_definitions(dubmist_host PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(dubmist_host PRIVATE juce::juce_audio_utils)
add_executable(dubmist_rhythm tests/rhythm.cpp)
target_link_libraries(dubmist_rhythm PRIVATE dubmist_core)
add_test(NAME rhythm_sample_timing COMMAND dubmist_rhythm)
